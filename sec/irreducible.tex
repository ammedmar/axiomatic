\section{Irreducibility}\label{s:irreducible}

In this section we introduce the notion of partition chains which form the largest subcomplex of irreducible chains.
We use them to prove that any irreducible semi-simplicial cup-\(i\) construction is simplicial.

\medskip A basis element $V \ot W \in \cP(\simplex^n)^{\ot 2}$ is said to be \textit{irreducible} if $V \cap W = \emptyset$ and \textit{reducible} otherwise.
Let $\pired,\pi^\perp_{\text{red}} \colon \cP(\simplex^n)^{\ot 2} \to \cP(\simplex^n)^{\ot 2}$ be the projections to the graded subspaces generated by reducible and irreducible basis elements respectively.
We remark that a cup-$i$ construction $\triangle$ is irreducible if and only if $\pired\triangle_i[n] = 0$ for all $i,n \in \N$.

The graded subspace of reducible chains is a subcomplex, i.e., it is closed under \(\partial\).
The graded subspace of irreducible chains is not.
We will now describe the largest subcomplex contained in it, the \textit{correflection} as it is sometimes called.

For any $U \subseteq \set{0,\dots,n}$ we define its \textit{partition chain} $\zeta_U$ as the sum of all basis elements $V \ot W$ in $\cP(\simplex^n)^{\ot 2}$ with $V \union W = U$ and $V \cap W = \emptyset$.

\begin{lemma}\label{l:partition chains}
	An irreducible chain $\zeta$ satisfies $\pired(\bd\zeta) = 0$ if and only if $\zeta$ is a sum of partition chains.
	Furthermore, if $\zeta_U$ is a partition chain then
	\begin{equation}\label{eq:boundary_partition_chain}
		\bd \zeta_U = \sum_{\bar u \notin U} \zeta_{\bar u.U}.
	\end{equation}
\end{lemma}

\begin{proof}
	Let $\Lambda$ be the set of distinct basis elements $V \ot W$ with $\zeta = \sum_\Lambda V \ot W$, then
	\[
	\bd \zeta =
	\sum_{\Lambda} \!\Big(\sum_{w \in W} w.V \ot W \ + \
	\sum_{v \in V} V \ot v.W \ + \
	\sum_{\mathclap{\bar u \notin V \union W}} \bar u.V \ot W + V \ot \bar u.W\Big).
	\]
	We then have
	\begin{align}
		\label{eq:non_perp}
		\pired(\bd \zeta) &=
		\sum_{\Lambda} \!\Big(\sum_{w \in W} w.V \ot W \ + \
		\sum_{v \in V} V \ot v.W\Big), \\
		\label{eq:perp}
		\pi^\perp_{\text{red}}(\bd \zeta) &=
		\sum_{\Lambda} \sum_{\quad\mathclap{\bar u \notin V \union W}} \bar u.V \ot W + V \ot \bar u.W.
	\end{align}
	The chain $\zeta$ is the partition chain of some $U$ if, by definition, $\Lambda$ contains all basis elements $V \ot W$ with $V \union W = U$ and $V \cap W = \emptyset$.
	In this case, \cref{eq:perp}, reads
	\[
	\pi^\perp_{\text{red}}(\bd \zeta) =
	\sum_{\Lambda} \sum_{\mathclap{\bar u \notin U}} \bar u.V \ot W + V \ot \bar u.W =
	\sum_{\quad\mathclap{\bar u \notin U}} \zeta_{\bar u.U}.
	\]
	This will then imply \cref{eq:boundary_partition_chain} after we show that $\zeta$ is in the kernel of $(\pired \circ \bd)$ when $\zeta = \zeta_U$.

	Let us now prove that the chain $\zeta$ satisfies $(\pired \circ \bd)\zeta = 0$ if and only if $\zeta$ is a sum of partition chains.
	Since for any two summands in \cref{eq:non_perp} we have
	\begin{align*}
		w.V \ot W = w'.V' \ot W' &\iff w = w', \ V = V', \ W = W', \\
		V \ot v.W = V' \ot v'.W' &\iff v = v', \ V = V', \ W = W',
	\end{align*}
	the sum of the elements in the following sets
	\begin{align*}
		L &= \set[\Big]{w.V \ot W \mid V \ot W \in \Lambda, \ w \in W}, \\
		R &= \set[\Big]{V \ot v.W \mid V \ot W \in \Lambda, \ v \in V},
	\end{align*}
	is equal to $(\pired \circ \bd) \zeta$.
	Therefore, $(\pired \circ \bd) \zeta = 0$ if and only $L = R$.
	Notice that for elements in $L$ and $R$ we have
	\begin{align*}
		w.V \ot W = V' \ot v'.W' &\iff v' = w, \ V' = w.V, \ W' = W \setminus w, \\
		V \ot v.W = w'.V' \ot W' &\iff w' = v, \ V' = V \setminus v, \ W' = v.W.
	\end{align*}

	These relations motivate the following notion.
	A \textit{basic move} on a partition $(V,W)$ removes an element from one of the factors and adds it to the other.
	Clearly, any two partitions of the same set are related by a sequence of basic moves.
	Therefore, $L = R$ if and only if for all $V \ot W \in \Lambda$ and any partition of $(V',W')$ obtained from $(V,W)$ via a basic move we have $V' \ot W' \in \Lambda$.
	In other words, $L = R$ if and only if $\zeta$ is a sum of partition chains.
\end{proof}

\begin{theorem}\label{t:irreducible_implies_simplicial}
	Every irreducible semi-simplicial \mbox{cup-$i$} construction is simplicial.
\end{theorem}

\begin{proof}
	Let $\triangle$ be an irreducible semi-simplicial \mbox{cup-$i$} construction, and fix $i, n \in \N$ with $n > 0$.
	We use the coefficient notation $\coeff{\zeta}{V \ot W}$ introduced in \cref{ss:axioms_revisited}.

	We first claim that
	\begin{equation}\label{eq:reducible_part}
		\pired\big(\bd\triangle_i[n]\big) = \triangle_i \bd\, [n].
	\end{equation}
	Indeed, $\triangle_i \bd\, [n] = \sum_{k=0}^{n} \cP(\delta_k)^{\ot 2}\, \triangle_i[n-1]$ and, since every subset in the image of $\cP(\delta_k)$ contains $k$, every basis element in the image of $\cP(\delta_k)^{\ot 2}$ is reducible.
	On the other hand, $(1+T)\triangle_{i-1}[n]$ is a sum of irreducible basis elements.
	Applying $\pired$ to the identity $\bd\triangle_i[n] + \triangle_i\bd\,[n] = (1+T)\triangle_{i-1}[n]$ proves \eqref{eq:reducible_part}.

	Let us compare coefficients on both sides of \eqref{eq:reducible_part}.
	Writing $\triangle_i[n] = \sum_\Lambda V \ot W$, we have, as in the proof of \cref{l:partition chains},
	\[
	\pired\big(\bd\triangle_i[n]\big) =
	\sum_{\Lambda}\Big(\sum_{w \in W} w.V \ot W \ + \ \sum_{v \in V} V \ot v.W\Big),
	\]
	so every basis element appearing on either side of \cref{eq:reducible_part} is of the form $X \ot Y$ with $X \cap Y = \set{k}$ for exactly one $k$.
	Fix such an $X \ot Y$.
	Its coefficient on the left hand side is
	\[
	\coeff{\triangle_i[n]}{X \setminus k \ot Y} + \coeff{\triangle_i[n]}{X \ot Y \setminus k}.
	\]
	For the right hand side, notice that $\cP(\delta_m)^{\ot 2}(A \ot B) = X \ot Y$ forces $m = k$, and that $\cP(\delta_k)$ defines a bijection from the subsets of $\set{0, \dots, n-1}$ to the subsets of $\set{0, \dots, n}$ containing $k$, with inverse $S \mapsto \sigma_k(S \setminus k)$.
	Therefore, \cref{eq:reducible_part} is equivalent to the identity
	\begin{equation}\label{eq:coefficient_comparison}
		\coeff{\triangle_i[n]}{X \setminus k \ot Y} + \coeff{\triangle_i[n]}{X \ot Y \setminus k} =
		\coeff{\triangle_i[n-1]}{\sigma_k(X \setminus k) \ot \sigma_k(Y \setminus k)}
	\end{equation}
	holding for every basis element $X \ot Y$ with $X \cap Y = \set{k}$.

	By \cref{l:simplicial_from_semisimplicial}, it suffices to show that $\cP(\sigma_j)^{\ot 2}\triangle_i[n] = 0$ for every codegeneracy $\sigma_j \colon [n] \to [n-1]$.
	Inspecting \cref{eq:codegeneracies}, an irreducible basis element $V \ot W$ satisfies $\cP(\sigma_j)^{\ot 2}(V \ot W) \neq 0$ if and only if one of $j, j+1$ belongs to $V$ and the other to $W$.
	For such an element, let $V' \ot W'$ be the irreducible basis element obtained by exchanging $j$ and $j+1$; explicitly, if $j \in V$ and $j+1 \in W$, then $V' = (V \setminus j) \union \set{j{+}1}$ and $W' = (W \setminus j{+}1) \union \set{j}$.
	A direct verification using \cref{eq:codegeneracies} shows that
	\[
	\cP(\sigma_j)^{\ot 2}(V \ot W) = \cP(\sigma_j)^{\ot 2}(V' \ot W')
	\]
	and that $V \ot W$ and $V' \ot W'$ are the only irreducible basis elements not annihilated by $\cP(\sigma_j)^{\ot 2}$ with this image.
	Consequently, since $\triangle_i[n]$ is irreducible, $\cP(\sigma_j)^{\ot 2}\, \triangle_i[n] = 0$ if and only if
	\begin{equation}\label{eq:swap}
		\coeff{\triangle_i[n]}{V \ot W} = \coeff{\triangle_i[n]}{V' \ot W'}
	\end{equation}
	for every such pair.

	To establish \cref{eq:swap}, assume $j \in V$ and $j+1 \in W$, the other case being symmetric.
	Applying \cref{eq:coefficient_comparison} to the basis element $V \ot \big(W \union \set{j}\big)$, whose factors intersect in $\set{j}$, we obtain
	\[
	\coeff{\triangle_i[n]}{V \setminus j \ot W \union \set{j}} + \coeff{\triangle_i[n]}{V \ot W} =
	\coeff{\triangle_i[n-1]}{\sigma_j(V \setminus j) \ot \sigma_j(W)},
	\]
	and applying it to $V' \ot \big(W \union \set{j}\big)$, whose factors intersect in $\set{j+1}$, we obtain
	\begin{multline}
		\coeff{\triangle_i[n]}{V \setminus j \ot W \union \set{j}} + \coeff{\triangle_i[n]}{V' \ot W'} = \\
		\coeff{\triangle_i[n-1]}{\sigma_{j+1}(V \setminus j) \ot \sigma_{j+1}\big((W \setminus j{+}1) \union \set{j}\big)}.
	\end{multline}
	Since $V \setminus j$ contains neither $j$ nor $j+1$, we have $\sigma_j(V \setminus j) = \sigma_{j+1}(V \setminus j)$, and a direct inspection gives $\sigma_j(W) = \sigma_{j+1}\big((W \setminus j{+}1) \union \set{j}\big)$.
	The right hand sides of the two identities above therefore agree, and adding them yields \cref{eq:swap}.
\end{proof}
